# AIBoxing 파일 구조

```text
AIBoxing/
├── README.md
├── pose_landmarker_heavy.task
├── requirements/
│   ├── base.txt              # 공통 패키지
│   ├── cpu.txt               # CPU 실행 환경
│   ├── amd.txt               # Windows AMD/Intel DirectML 환경
│   ├── cuda11.txt            # NVIDIA CUDA 11 환경
│   └── cuda12.txt            # NVIDIA CUDA 12 환경
└── src/
    ├── config.py             # 공통 경로 및 장치 선택
    ├── data/
    │   ├── images/
    │   ├── videos/
    │   ├── annotations.json
    │   └── processed_videos.json
    ├── models/pose_net.py     # 3D 포즈 모델
    ├── utils/
    │   ├── dataset.py
    │   └── loss.py
    ├── weights/                # .pth/.onnx 모델 파일
    ├── output/                 # 추론 결과 영상
    ├── label.py                # MediaPipe 라벨링
    ├── train.py                # 모델 학습
    ├── export.py               # PyTorch -> ONNX 변환
    ├── inference.py            # 영상/웹캠 추론
    └── merge_weights.py        # 체크포인트 평균 병합
```
